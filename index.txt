    <script>
        const CURRENT_YEAR = 2026;
        const periods = ["1교시", "2교시", "3교시", "4교시", "5교시", "6교시", "7교시"];
        const periodTimes = ["09:00~09:40", "09:50~10:30", "10:40~11:20", "11:30~12:10", "12:20~13:00", "13:00~13:40", "13:50~14:30"];
        
        let currentDate = new Date();
        let currentSelectedPeriod = null;
        let isEditMode = false;

        let reservationsData = JSON.parse(localStorage.getItem('gymData')) || {};
        let adminPassword = localStorage.getItem('gymPwd') || "0000";

        function saveData() {
            localStorage.setItem('gymData', JSON.stringify(reservationsData));
        }

        function showScreen(screenId) {
            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
            document.getElementById(screenId).classList.add('active');
            if(screenId === 'screen-month-select') renderMonthSelect();
        }

        function goHome() {
            showScreen('screen-home');
        }

        function formatDateStr(date) {
            const days = ["일", "월", "화", "수", "목", "금", "토"];
            return `${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}-${days[date.getDay()]}요일`;
        }
        
        function getDateKey(date) {
            return `${CURRENT_YEAR}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`;
        }

        function goToToday() {
            currentDate = new Date();
            updateDailyDisplay();
            showScreen('screen-daily');
        }

        function renderMonthSelect() {
            const grid = document.getElementById("monthGrid");
            grid.innerHTML = "";
            for(let i=1; i<=12; i++) {
                let btn = document.createElement("div");
                btn.className = "month-btn";
                btn.innerText = `${i}월`;
                btn.onclick = () => renderCalendar(i);
                grid.appendChild(btn);
            }
        }

        function renderCalendar(month) {
            document.getElementById("calendarTitle").innerText = `${CURRENT_YEAR}년 ${month}월`;
            const grid = document.getElementById("calendarGrid");
            grid.innerHTML = "";
            
            const days = ["일", "월", "화", "수", "목", "금", "토"];
            days.forEach(d => {
                grid.innerHTML += `<div class="cal-header">${d}</div>`;
            });

            const firstDay = new Date(CURRENT_YEAR, month - 1, 1).getDay();
            const lastDate = new Date(CURRENT_YEAR, month, 0).getDate();

            for(let i=0; i<firstDay; i++) { grid.innerHTML += `<div class="cal-day empty"></div>`; }

            for(let day=1; day<=lastDate; day++) {
                let targetDate = new Date(CURRENT_YEAR, month - 1, day);
                let dateKey = getDateKey(targetDate);
                let dayOfWeek = targetDate.getDay();
                
                let dayDiv = document.createElement("div");
                dayDiv.innerText = day;
                
                if (dayOfWeek === 0 || dayOfWeek === 6) {
                    dayDiv.className = "cal-day weekend";
                } else {
                    let dayData = reservationsData[dateKey] || {};
                    let bookedCount = Object.keys(dayData).length;
                    
                    if (bookedCount >= 7) {
                        dayDiv.className = "cal-day full";
                    } else {
                        dayDiv.className = "cal-day available";
                    }
                    
                    dayDiv.onclick = () => {
                        currentDate = targetDate;
                        updateDailyDisplay();
                        showScreen('screen-daily');
                    };
                }
                grid.appendChild(dayDiv);
            }
            showScreen('screen-calendar');
        }

        function changeDate(days) {
            currentDate.setDate(currentDate.getDate() + days);
            updateDailyDisplay();
        }

        function updateDailyDisplay() {
            document.getElementById("currentDateDisplay").innerText = formatDateStr(currentDate);
            const tbody = document.getElementById("tableBody");
            tbody.innerHTML = "";
            const dateKey = getDateKey(currentDate);
            const dayData = reservationsData[dateKey] || {};

            periods.forEach((periodName, index) => {
                const tr = document.createElement("tr");
                const res = dayData[periodName];
                
                if (res) {
                    tr.innerHTML = `
                        <td><div class="period-info">${periodName}</div><div class="time-info">${periodTimes[index]}</div></td>
                        <td>${res.text} <span class="author-text">${res.author}</span></td>
                        <td><span class="status-red">예약 불가</span></td>
                    `;
                } else {
                    tr.innerHTML = `
                        <td><div class="period-info">${periodName}</div><div class="time-info">${periodTimes[index]}</div></td>
                        <td class="slot-available" onclick="openReservationModal('${periodName}')"><span style="color:#aaa;">(클릭하여 예약)</span></td>
                        <td><span class="status-blue">예약 가능</span></td>
                    `;
                }
                tbody.appendChild(tr);
            });
        }

        function openReservationModal(period, dateKey = null) {
            isEditMode = dateKey !== null;
            currentSelectedPeriod = period;
            
            if (isEditMode) {
                document.getElementById("modalTitle").innerText = "예약 수정";
                document.getElementById("editDateKey").value = dateKey;
            } else {
                document.getElementById("modalTitle").innerText = `${formatDateStr(currentDate)} ${period} 예약`;
                document.getElementById("editDateKey").value = getDateKey(currentDate);
            }
            
            // 모달창 열 때 초기화 처리
            document.getElementById("resGrade").value = "";
            document.getElementById("resClass").value = "";
            document.getElementById("resTeacher").value = "";
            document.getElementById("reservationModal").style.display = "flex";
        }

        function closeModal(id) {
            document.getElementById(id).style.display = "none";
        }

        function saveReservation() {
            const dateKey = document.getElementById("editDateKey").value;
            const grade = document.getElementById("resGrade").value || '';
            const classNum = document.getElementById("resClass").value || '';
            const teacher = document.getElementById("resTeacher").value;

            if (!teacher) { alert("선생님 성함이나 내용을 입력해주세요."); return; }

            // 학년이나 반을 선택하지 않고 선생님 성함만 적었을 경우도 고려해 trim()으로 깔끔하게 정리
            const text = `${grade} ${classNum} ${teacher}`.trim();
            
            if (!reservationsData[dateKey]) reservationsData[dateKey] = {};
            // 작성자 이름이 없어졌으므로 선생님 성함을 작성자로 대신 표시합니다.
            reservationsData[dateKey][currentSelectedPeriod] = { text: text, author: `작성자: ${teacher}` };
            
            saveData();
            closeModal('reservationModal');
            
            alert("저장되었습니다.");
            
            if (isEditMode) {
                openAdminList();
            } else {
                updateDailyDisplay();
            }
        }

        function openAdminModal() {
            document.getElementById("adminLogin").style.display = "block";
            document.getElementById("adminPanel").style.display = "none";
            document.getElementById("adminPwdInput").value = "";
            document.getElementById("adminModal").style.display = "flex";
        }

        function checkAdmin() {
            if (document.getElementById("adminPwdInput").value === adminPassword) {
                document.getElementById("adminLogin").style.display = "none";
                document.getElementById("adminPanel").style.display = "block";
            } else {
                alert("비밀번호가 틀렸습니다.");
            }
        }

        function openChangePwdModal() {
            closeModal('adminModal');
            document.getElementById("oldPwd").value = "";
            document.getElementById("newPwd").value = "";
            document.getElementById("newPwdConfirm").value = "";
            document.getElementById("pwdModal").style.display = "flex";
        }

        function saveNewPassword() {
            const old = document.getElementById("oldPwd").value;
            const newP = document.getElementById("newPwd").value;
            const confirmP = document.getElementById("newPwdConfirm").value;

            if (old !== adminPassword) { alert("현재 비밀번호가 틀립니다."); return; }
            if (newP !== confirmP) { alert("바꿀 비밀번호가 서로 일치하지 않습니다."); return; }
            if (newP === "") { alert("새 비밀번호를 입력하세요."); return; }

            localStorage.setItem('gymPwd', newP);
            alert("비밀번호가 성공적으로 변경되었습니다.");
            closeModal('pwdModal');
        }

        function openAdminList() {
            closeModal('adminModal');
            const tbody = document.getElementById("adminListBody");
            tbody.innerHTML = "";

            const sortedDates = Object.keys(reservationsData).sort();
            
            sortedDates.forEach(dateKey => {
                const dayData = reservationsData[dateKey];
                periods.forEach(period => {
                    if (dayData[period]) {
                        const tr = document.createElement("tr");
                        tr.innerHTML = `
                            <td>${dateKey}</td>
                            <td>${period}</td>
                            <td>${dayData[period].text}</td>
                            <td>
                                <button class="btn-action btn-edit" onclick="openReservationModal('${period}', '${dateKey}')">수정</button>
                                <button class="btn-action btn-delete" onclick="deleteReservation('${dateKey}', '${period}')">삭제</button>
                            </td>
                        `;
                        tbody.appendChild(tr);
                    }
                });
            });
            showScreen('screen-admin-list');
        }

        function deleteReservation(dateKey, period) {
            if (confirm(`정말 ${dateKey} ${period} 예약을 삭제하시겠습니까?`)) {
                delete reservationsData[dateKey][period];
                if (Object.keys(reservationsData[dateKey]).length === 0) {
                    delete reservationsData[dateKey];
                }
                saveData();
                alert("삭제되었습니다.");
                openAdminList();
            }
        }
